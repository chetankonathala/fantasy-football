    **Step 1 — Create SearchBar.tsx:**

    Create `frontend/src/components/SearchBar.tsx`:

    ```typescript
    "use client";

    import { useState, useEffect, useRef, useCallback } from "react";
    import { useRouter } from "next/navigation";
    import { useDebounce } from "use-debounce";
    import { Search, Loader2 } from "lucide-react";

    interface PlayerResult {
      id: number;
      full_name: string;
      position: string;
      team: string;
    }

    const API_BASE = process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000";

    export function SearchBar() {
      const [query, setQuery] = useState("");
      const [results, setResults] = useState<PlayerResult[]>([]);
      const [isLoading, setIsLoading] = useState(false);
      const [isOpen, setIsOpen] = useState(false);
      const [activeIndex, setActiveIndex] = useState(-1);
      const [debouncedQuery] = useDebounce(query, 300);
      const router = useRouter();
      const inputRef = useRef<HTMLInputElement>(null);
      const listRef = useRef<HTMLUListElement>(null);

      useEffect(() => {
        if (debouncedQuery.length < 2) {
          setResults([]);
          setIsOpen(false);
          return;
        }
        setIsLoading(true);
        fetch(`${API_BASE}/search?q=${encodeURIComponent(debouncedQuery)}`)
          .then((r) => r.json())
          .then((data: PlayerResult[]) => {
            setResults(data);
            setIsOpen(true);
            setActiveIndex(-1);
          })
          .catch(() => {
            setResults([]);
            setIsOpen(true);
          })
          .finally(() => setIsLoading(false));
      }, [debouncedQuery]);

      const selectPlayer = useCallback(
        (player: PlayerResult) => {
          router.push(`/player/${player.id}`);
          setQuery("");
          setResults([]);
          setIsOpen(false);
        },
        [router]
      );

      const handleKeyDown = useCallback(
        (e: React.KeyboardEvent) => {
          if (!isOpen) return;
          if (e.key === "ArrowDown") {
            e.preventDefault();
            setActiveIndex((prev) => Math.min(prev + 1, results.length - 1));
          } else if (e.key === "ArrowUp") {
            e.preventDefault();
            setActiveIndex((prev) => Math.max(prev - 1, -1));
          } else if (e.key === "Enter" && activeIndex >= 0) {
            e.preventDefault();
            selectPlayer(results[activeIndex]);
          } else if (e.key === "Escape") {
            setIsOpen(false);
            setActiveIndex(-1);
          }
        },
        [isOpen, activeIndex, results, selectPlayer]
      );

      const showDropdown = isOpen && debouncedQuery.length >= 2;
      const showEmpty = showDropdown && results.length === 0 && !isLoading;
      const showResults = showDropdown && results.length > 0;
      const showLoading = showDropdown && isLoading && results.length === 0;

      return (
        <div className="relative">
          <div className="relative">
            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-eagles-silver" />
            <input
              ref={inputRef}
              value={query}
              onChange={(e) => setQuery(e.target.value)}
              onKeyDown={handleKeyDown}
              onFocus={() => {
                if (results.length > 0 && debouncedQuery.length >= 2) setIsOpen(true);
              }}
              onBlur={() => setTimeout(() => setIsOpen(false), 200)}
              placeholder="Search players..."
              aria-label="Search players"
              role="combobox"
              aria-expanded={showDropdown}
              aria-controls="search-listbox"
              aria-activedescendant={activeIndex >= 0 ? `search-option-${activeIndex}` : undefined}
              className="w-full bg-surface-card border border-eagles-silver/30 rounded-lg pl-9 pr-9 py-2 text-white placeholder:text-zinc-500 focus:outline-none focus:border-eagles-green focus:border-2 text-sm"
            />
            {isLoading && (
              <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-eagles-silver animate-spin" />
            )}
          </div>

          {showResults && (
            <ul
              ref={listRef}
              id="search-listbox"
              role="listbox"
              className="absolute w-full bg-surface-card border border-eagles-silver/20 rounded-md mt-1 z-50 shadow-lg overflow-hidden"
            >
              {results.map((p, i) => (
                <li
                  key={p.id}
                  id={`search-option-${i}`}
                  role="option"
                  aria-selected={i === activeIndex}
                  onMouseDown={() => selectPlayer(p)}
                  onMouseEnter={() => setActiveIndex(i)}
                  className={`px-4 min-h-[44px] flex items-center cursor-pointer text-white text-base transition-colors ${
                    i === activeIndex ? "bg-eagles-green/50" : "hover:bg-eagles-green/30"
                  }`}
                >
                  {p.full_name} — {p.position}, {p.team}
                </li>
              ))}
            </ul>
          )}

          {showLoading && (
            <div
              id="search-listbox"
              role="listbox"
              className="absolute w-full bg-surface-card border border-eagles-silver/20 rounded-md mt-1 z-50 shadow-lg"
            >
              <div className="px-4 py-3 min-h-[44px] flex items-center">
                <div className="h-4 w-48 bg-gray-700 rounded animate-pulse" />
              </div>
            </div>
          )}

          {showEmpty && (
            <div className="absolute w-full bg-surface-card border border-eagles-silver/20 rounded-md mt-1 z-50 shadow-lg px-4 py-3">
              <p className="text-eagles-silver text-sm font-medium">No players found</p>
              <p className="text-zinc-500 text-xs mt-1">Try a different name or check the spelling.</p>
            </div>
          )}
        </div>
      );
    }
    ```

    Key UI-SPEC.md requirements implemented:
    - Trigger: 2+ characters before any API call
    - Debounce: 300ms via use-debounce
    - Dropdown: z-50, bg surface-card (#111827), rounded-md, shadow-lg
    - Suggestion item: 44px min-height, 16px horizontal padding, 8px vertical padding
    - Suggestion format: "{full_name} -- {position}, {team}"
    - Max 5 items (enforced by API limit)
    - Empty state: "No players found" + "Try a different name or check the spelling."
    - Loading state: single animated pulse skeleton
    - Keyboard: ArrowDown/Up navigate, Enter selects, Escape closes
    - Accessibility: role="combobox", aria-expanded, role="listbox", role="option", aria-selected
    - Focus border: eagles-green (#004C54) 2px
    - Hover: eagles-green at 30% opacity
    - Keyboard focus: eagles-green at 50% opacity
    - Search icon (lucide-react Search) and loading spinner (Loader2)
