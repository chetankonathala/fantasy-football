    ```typescript
    "use client";

    interface FreshnessStampProps {
      updatedAt: string; // ISO 8601 string
    }

    function relativeTime(isoString: string): string {
      const diff = Date.now() - new Date(isoString).getTime();
      const seconds = Math.floor(diff / 1000);
      if (seconds < 60) return "Updated just now";
      const minutes = Math.floor(seconds / 60);
      if (minutes < 60) return `Updated ${minutes} minute${minutes !== 1 ? "s" : ""} ago`;
      const hours = Math.floor(minutes / 60);
      if (hours < 24) return `Updated ${hours} hour${hours !== 1 ? "s" : ""} ago`;
      const days = Math.floor(hours / 24);
      return `Updated ${days} day${days !== 1 ? "s" : ""} ago`;
    }

    export function FreshnessStamp({ updatedAt }: FreshnessStampProps) {
      return (
        <p className="text-sm text-eagles-silver mt-8 pt-2">
          {relativeTime(updatedAt)}
        </p>
      );
    }
    ```

    Thresholds per UI-SPEC.md:
    - < 60s: "Updated just now"
    - < 60m: "Updated N minutes ago"
    - < 24h: "Updated N hours ago"
    - else: "Updated N days ago"

    ```typescript
    "use client";

    import { useState, useEffect } from "react";
    import { FreshnessStamp } from "./FreshnessStamp";

    interface RecommendationResponse {
      verdict: "START" | "SIT" | "FLEX";
      score: number;
      reasons: string[];
      low_confidence: boolean;
      scoring_format: "ppr" | "half_ppr" | "standard";
      injury_status: string | null;
      practice_participation: string | null;
      snap_pct_l4w: (number | null)[];
      target_share_l4w: (number | null)[];
      carry_share_l4w: (number | null)[];
      updated_at: string;
      full_name: string;
      position: string;
      team: string;
      projected_points: number | null;
    }

    type Format = "ppr" | "half_ppr" | "standard";

    const FORMAT_LABELS: Record<Format, string> = {
      ppr: "PPR",
      half_ppr: "Half PPR",
      standard: "Standard",
    };

    const VERDICT_COLORS: Record<string, string> = {
      START: "text-verdict-start",
      SIT: "text-verdict-sit",
      FLEX: "text-verdict-flex",
    };

    const API_BASE = process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000";

    function InjuryBadge({ status }: { status: string | null }) {
      if (!status || status === "") {
        return <span className="text-verdict-start bg-green-950 px-2 py-0.5 rounded text-sm">Active</span>;
      }
      const map: Record<string, { text: string; bg: string }> = {
        Questionable: { text: "text-verdict-flex", bg: "bg-[#1C1400]" },
        Doubtful: { text: "text-red-400", bg: "bg-[#1A0000]" },
        Out: { text: "text-verdict-sit", bg: "bg-[#1A0000]" },
      };
      const style = map[status] || { text: "text-white", bg: "bg-gray-800" };
      return (
        <span className={`${style.text} ${style.bg} px-2 py-0.5 rounded text-sm`}>
          {status}
        </span>
      );
    }

    function WeeklyValues({ values }: { values: (number | null)[] }) {
      return (
        <span className="text-white text-base">
          {values.map((v, i) => (
            <span key={i}>
              {v !== null ? `${(v * 100).toFixed(0)}%` : "—"}
              {i < values.length - 1 && <span className="text-eagles-silver mx-1">·</span>}
            </span>
          ))}
        </span>
      );
    }

    interface RecommendationCardProps {
      playerId: string;
      initialData: RecommendationResponse;
    }

    export function RecommendationCard({ playerId, initialData }: RecommendationCardProps) {
      const [format, setFormat] = useState<Format>(
        (initialData.scoring_format as Format) || "ppr"
      );
      const [rec, setRec] = useState<RecommendationResponse>(initialData);
      const [isLoading, setIsLoading] = useState(false);
      const [error, setError] = useState<string | null>(null);

      useEffect(() => {
        // Skip fetch if format matches initial data (first render)
        if (format === initialData.scoring_format) {
          setRec(initialData);
          return;
        }
        setIsLoading(true);
        setError(null);
        fetch(`${API_BASE}/player/${playerId}?format=${format}`)
          .then((r) => {
            if (!r.ok) throw new Error("Failed to fetch");
            return r.json();
          })
          .then((data: RecommendationResponse) => {
            setRec(data);
          })
          .catch(() => {
            setError("Could not load recommendation");
          })
          .finally(() => setIsLoading(false));
      }, [format, playerId, initialData]);

      // Loading skeleton
      if (isLoading) {
        return (
          <div
            className="bg-surface-card rounded-xl p-6"
            aria-busy="true"
            aria-label="Loading recommendation..."
          >
            <div className="h-9 w-32 bg-gray-700 rounded animate-pulse mb-4" />
            <div className="h-4 w-24 bg-gray-700 rounded animate-pulse mb-6" />
            <div className="flex gap-2 mb-6">
              {[1, 2, 3].map((i) => (
                <div key={i} className="h-8 w-20 bg-gray-700 rounded animate-pulse" />
              ))}
            </div>
            <div className="space-y-3">
              {[1, 2, 3, 4].map((i) => (
                <div key={i} className="h-4 w-full bg-gray-700 rounded animate-pulse" />
              ))}
            </div>
            <div className="mt-6 space-y-3">
              {[1, 2, 3, 4, 5, 6].map((i) => (
                <div key={i} className="h-4 w-3/4 bg-gray-700 rounded animate-pulse" />
              ))}
            </div>
          </div>
        );
      }

      // Error state
      if (error) {
        return (
          <div className="bg-surface-card rounded-xl p-6" role="alert">
            <h3 className="text-white text-xl font-bold mb-2">Could not load recommendation</h3>
            <p className="text-eagles-silver text-base mb-4">
              There was a problem fetching data for this player. Retry loading or search for another player.
            </p>
            <button
              onClick={() => {
                setError(null);
                setIsLoading(true);
                fetch(`${API_BASE}/player/${playerId}?format=${format}`)
                  .then((r) => r.json())
                  .then(setRec)
                  .catch(() => setError("Could not load recommendation"))
                  .finally(() => setIsLoading(false));
              }}
              className="bg-eagles-green text-white px-4 py-2 rounded-lg text-sm font-bold hover:bg-eagles-green/80"
            >
              Retry Loading
            </button>
          </div>
        );
      }

      // Determine which usage metric to show based on position
      const showTargetShare = rec.position === "WR" || rec.position === "TE";
      const showCarryShare = rec.position === "RB";
      const usageLabel = showTargetShare
        ? "Target Share (L4W)"
        : showCarryShare
        ? "Carry Share (L4W)"
        : null;
      const usageValues = showTargetShare
        ? rec.target_share_l4w
        : showCarryShare
        ? rec.carry_share_l4w
        : null;

      return (
        <div className="bg-surface-card rounded-xl p-6">
          {/* Verdict block */}
          <div className="flex items-baseline justify-between">
            <span
              className={`text-4xl font-bold ${VERDICT_COLORS[rec.verdict]}`}
              aria-label={`Recommendation: ${rec.verdict}`}
            >
              {rec.verdict}
            </span>
            <span className="text-sm text-eagles-silver">
              Score: {rec.score.toFixed(1)}
            </span>
          </div>

          {/* Low confidence badge */}
          {rec.low_confidence && (
            <div className="mt-2">
              <span className="text-sm text-verdict-flex bg-eagles-silver/20 px-2 py-0.5 rounded">
                Low confidence — limited data
              </span>
            </div>
          )}

          {/* Scoring format selector */}
          <div className="flex gap-2 mt-4">
            {(["ppr", "half_ppr", "standard"] as Format[]).map((f) => (
              <button
                key={f}
                onClick={() => setFormat(f)}
                className={`px-3 h-8 rounded text-sm min-h-[44px] transition-colors ${
                  format === f
                    ? "bg-eagles-green text-white font-bold"
                    : "bg-transparent border border-eagles-silver text-eagles-silver hover:bg-eagles-green/20"
                }`}
              >
                {FORMAT_LABELS[f]}
              </button>
            ))}
          </div>

          {/* Reasoning block */}
          <div className="mt-6">
            <h3 className="text-sm text-eagles-silver mb-3">Why</h3>
            <ul className="space-y-2">
              {rec.reasons.map((reason, i) => (
                <li key={i} className="flex items-start gap-2 text-base text-white">
                  <span className="text-eagles-silver mt-0.5">·</span>
                  <span>{reason}</span>
                </li>
              ))}
            </ul>
          </div>

          {/* Signals block */}
          <div className="mt-6">
            <h3 className="text-sm text-eagles-silver mb-3">Signals</h3>
            <div className="space-y-3">
              {/* Matchup Grade */}
              <div className="flex justify-between items-center">
                <span className="text-sm text-eagles-silver">Matchup Grade</span>
                <span className="text-base text-white">
                  {rec.score >= 70 ? "A" : rec.score >= 55 ? "B" : rec.score >= 40 ? "C" : "D"}
                </span>
              </div>

              {/* Injury Status */}
              <div className="flex justify-between items-center">
                <span className="text-sm text-eagles-silver">Injury Status</span>
                <InjuryBadge status={rec.injury_status} />
              </div>

              {/* Practice Status */}
              <div className="flex justify-between items-center">
                <span className="text-sm text-eagles-silver">Practice Status</span>
                <span className="text-base text-white">
                  {rec.practice_participation || "Full"}
                </span>
              </div>

              {/* Snap % L4W */}
              <div className="flex justify-between items-center">
                <span className="text-sm text-eagles-silver">Snap % (L4W)</span>
                <WeeklyValues values={rec.snap_pct_l4w} />
              </div>

              {/* Target/Carry Share L4W (position-gated) */}
              {usageLabel && usageValues && (
                <div className="flex justify-between items-center">
                  <span className="text-sm text-eagles-silver">{usageLabel}</span>
                  <WeeklyValues values={usageValues} />
                </div>
              )}

              {/* Projected Points */}
              <div className="flex justify-between items-center">
                <span className="text-sm text-eagles-silver">Projected Points</span>
                <span className="text-base text-white">
                  {rec.projected_points !== null ? `${rec.projected_points.toFixed(1)} pts` : "—"}
                </span>
              </div>
            </div>
          </div>

          {/* Freshness stamp */}
          <FreshnessStamp updatedAt={rec.updated_at} />
        </div>
      );
    }
    ```

    ```typescript
    import { notFound } from "next/navigation";

    const API_BASE = process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000";

    interface RecommendationResponse {
      verdict: "START" | "SIT" | "FLEX";
      score: number;
      reasons: string[];
      low_confidence: boolean;
      scoring_format: "ppr" | "half_ppr" | "standard";
      injury_status: string | null;
      practice_participation: string | null;
      snap_pct_l4w: (number | null)[];

      carry_share_l4w: (number | null)[];
      updated_at: string;
      full_name: string;
      position: string;
      team: string;
      projected_points: number | null;
    }

    async function getPlayerRecommendation(id: string): Promise<RecommendationResponse | null> {
      try {
        const res = await fetch(`${API_BASE}/player/${id}?format=ppr`, {
          cache: "no-store",
        });
        if (!res.ok) return null;
        return res.json();
      } catch {
        return null;
      }
    }

    export default async function PlayerPage({
      params,
    }: {
      params: Promise<{ id: string }>;
    }) {
      const { id } = await params;
      const data = await getPlayerRecommendation(id);

      if (!data) {
        notFound();
      }

      return (
        <div className="max-w-2xl mx-auto px-8 py-8">
          {/* Player header */}
          <div className="mb-6">
            <h1 className="text-xl font-bold text-white">{data.full_name}</h1>
            <p className="text-sm text-eagles-silver mt-1">
              {data.position} · {data.team}
            </p>
          </div>

          {/* Recommendation card */}
          <RecommendationCard playerId={id} initialData={data} />
        </div>
      );
    }
    ```
